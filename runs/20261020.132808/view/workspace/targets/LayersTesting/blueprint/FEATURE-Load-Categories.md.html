Consumescategories.get, categories.create, categories.list

Purpose

Loads the categories table from a CSV file whose header is category,description.

Behavior

  • The header must equal category,description in that order. Otherwise the command exits 3 and stores nothing.
  • Each code is validated by the code rules. Each description is 1 to 40 characters with no comma or newline. A violation exits 3 and stores nothing.
  • A code repeated within the file, or a code already present, exits 3 and stores nothing.
  • A header-only file loads zero rows and exits 0.
  • On success, stdout is table,loaded then categories,<rows>.
  • The command runs in one command transaction. An unreadable file exits 1 with no change.

import csv import io import os import sys import tempfile

from cli.testing import cli_runner

os.environ["DATABASE_URL"] = os.path.join(tempfile.mkdtemp(), "ledger.db")

def run(*args): code, out, err = cli_runner(*args) print(out) print(err, file=sys.stderr) return code, out, err

seed_path = "sources/seed_categories.csv" with open(seed_path, newline="") as handle: seed_header, *seed_rows = list(csv.reader(handle)) assert seed_header == ["category", "description"]

assert run("init")[0] == 0 code, out, err = run("load", "categories", seed_path) assert code == 0 loaded = list(csv.reader(io.StringIO(out))) assert loaded[0] == ["table", "loaded"] assert loaded[-1] == ["categories", str(len(seed_rows))]

code, out, err = run("report", "categories") assert code == 0 report = {row[0]: row for row in list(csv.reader(io.StringIO(out)))[1:]} for category, description in seed_rows: assert report[category][1] == description === END AC load-categories-seed-round-trip ===

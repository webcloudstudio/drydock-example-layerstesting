FEATURE: Load Rejection

FieldValue
Version20261025 V1
DescriptionGuarantees that a rejected ledger load exits 3 and leaves every table exactly as it was.
Depends OnARCHITECTURE.md, DATABASE.md, FEATURE-LoadAccounts.md, FEATURE-LoadCategories.md, FEATURE-LoadTransactions.md, FEATURE-Count.md
Providesatomic load rejection
Consumesledger load accounts, ledger load categories, ledger load transactions, ledger count

Purpose

Prove and enforce the all-or-nothing rule for loads: a file with one invalid row stores none of its rows. A file of ten valid rows followed by an eleventh naming an unknown account stores none of the eleven.

Behavior

ScenarioResult
Transactions file whose last row names an unknown accountExit 3, nothing on stdout, no table changes.
Transactions file with an invalid amount, date, or duplicate id on a late rowExit 3, no table changes.
Accounts file reloaded after it was loadedExit 3, no table changes.
Categories file reloaded after it was loadedExit 3, no table changes.

Row counts of all three tables read through ledger count are identical before and after every rejected command.

Database Interface

  • FEATURE-TXN-002 — read: verify no transaction row was stored - supplies: account - returns: txn_id, account, amount
  • FEATURE-ACCOUNT-002 — read: verify no account row was stored - supplies: account - returns: account, name

=== AC load-rejection-eleventh-row ===
Intent: Ten valid rows followed by an unknown-account row are rejected with exit 3 and none of the eleven is stored.

import csv import io import os import tempfile from cli.testing import cli_runner from persistence.testing import db_session

def counts(): rc, out, err = cli_runner("count") assert rc == 0 return {r[0]: int(r[1]) for r in list(csv.reader(io.StringIO(out)))[1:]}

def load(table, header, rows, tmp, name=None): path = os.path.join(tmp, (name or table) + ".csv") with open(path, "w", newline="") as fh: w = csv.writer(fh, lineterminator="\n") w.writerow(header) w.writerows(rows) return cli_runner("load", table, path)

header = ["txn_id", "date", "account", "category", "amount", "memo"] acct, cat = "LDRJA1", "LDRJC1" with db_session() as db: base = max([t.txn_id for t in db.transactions.list()], default=0) + 1000 rows = [[base + i, "2026-05-01", acct, cat, "10.00", "batch %d" % i] for i in range(1, 11)] rows.append([base + 11, "2026-05-01", "LDRJNONE", cat, "10.00", "unknown account"]) with tempfile.TemporaryDirectory() as tmp: assert load("accounts", ["account", "name"], [[acct, "Reject Account"]], tmp)[0] == 0 assert load("categories", ["category", "description"], [[cat, "Reject Category"]], tmp)[0] == 0 before = counts() rc, out, err = load("transactions", header, rows, tmp, "eleven") print(out, err) assert rc == 3 assert out == "" assert counts() == before with db_session() as db: for i in range(1, 12): assert db.transactions.get(txn_id=base + i) is None === END AC load-rejection-eleventh-row ===

=== AC load-rejection-reload-existing ===
Intent: Reloading accounts or categories that already exist is rejected with exit 3 and changes no table.

import csv import io import os import tempfile from cli.testing import cli_runner

def counts(): rc, out, err = cli_runner("count") assert rc == 0 return {r[0]: int(r[1]) for r in list(csv.reader(io.StringIO(out)))[1:]}

def write(path, header, rows): with open(path, "w", newline="") as fh: w = csv.writer(fh, lineterminator="\n") w.writerow(header) w.writerows(rows)

with tempfile.TemporaryDirectory() as tmp: accounts = os.path.join(tmp, "accounts.csv") categories = os.path.join(tmp, "categories.csv") write(accounts, ["account", "name"], [["LDRRA1", "Reload A"], ["LDRRA2", "Reload B"]]) write(categories, ["category", "description"], [["LDRRC1", "Reload C"]]) assert cli_runner("load", "accounts", accounts)[0] == 0 assert cli_runner("load", "categories", categories)[0] == 0 before = counts() for table, path in (("accounts", accounts), ("categories", categories)): rc, out, err = cli_runner("load", table, path) print(table, out, err) assert rc == 3 assert out == "" assert counts() == before === END AC load-rejection-reload-existing ===

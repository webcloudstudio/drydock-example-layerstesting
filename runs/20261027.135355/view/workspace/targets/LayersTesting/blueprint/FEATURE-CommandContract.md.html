FieldValue
Version20261027 V1
DescriptionAssembly story verifying the CSV output contract and exit codes across every delivered ledger command.

Purpose

Assembly and intent only. Verify across the delivered commands that:

  • successful output is CSV with the specified headers, \n endings, two-decimal amounts, no quoting, and no trailing spaces;
  • a failing command writes nothing to standard output and one diagnostic line to standard error;
  • exit 2 results from unknown commands, tables, reports, and wrong argument counts;
  • exit 1 results from an unreadable file, 3 from rejection, 4 from not found;
  • every rejected writing command leaves the table counts unchanged.

Any seam defect is fixed in the owning module; this story adds no new behavior.

Database Interface

  • None — persistence is owned by DATABASE.md.

=== AC contract-csv-format ===
Intent: Successful commands emit the declared headers, newline-only endings, no trailing spaces, and amounts with exactly two decimals.

import csv import io import re from cli.testing import cli_runner from persistence.testing import db_session

def parse(out): return list(csv.reader(io.StringIO(out)))

def clean(out): assert out.endswith("\n") assert "\r" not in out for line in out.split("\n"): assert line == line.rstrip(" ")

acct = "ZZCF1" cat = "ZZCFC" with db_session() as db: if db.categories.get(category=cat) is None: db.categories.create(category=cat, description="Format Probe") code, out, err = cli_runner("account", "add", acct, "Format Probe") assert code == 0 clean(out) assert parse(out)[0] == ["account", "name", "transactions", "balance"]

code, out, err = cli_runner("txn", "post", acct, cat, "7.5", "2099-05-05", "fmt") print(out, err) assert code == 0 clean(out) table = parse(out) assert table[0] == ["txn_id", "date", "account", "category", "amount", "memo"] assert re.fullmatch(r"-?\d+\.\d{2}", table[1][4])

code, out, err = cli_runner("count") assert code == 0 clean(out) table = parse(out) assert table[0] == ["table", "rows"] assert [r[0] for r in table[1:]] == ["accounts", "categories", "transactions"]

code, out, err = cli_runner("report", "balances") assert code == 0 clean(out) assert parse(out)[0] == ["account", "name", "transactions", "balance"] code, out, err = cli_runner("report", "categories") assert code == 0 clean(out) assert parse(out)[0] == ["category", "description", "transactions", "inflow", "outflow", "net"] code, out, err = cli_runner("report", "monthly") assert code == 0 clean(out) assert parse(out)[0] == ["month", "transactions", "inflow", "outflow", "net"] === END AC contract-csv-format ===

=== AC contract-rejected-writes-change-nothing ===
Intent: Rejected writing commands exit 3 and leave the counts of all three tables unchanged.

import csv import io from cli.testing import cli_runner from persistence.testing import db_session

def counts(): code, out, err = cli_runner("count") assert code == 0 return {r[0]: int(r[1]) for r in list(csv.reader(io.StringIO(out)))[1:]}

acct = "ZZCR1" other = "ZZCR2" cat = "ZZCRC" with db_session() as db: if db.categories.get(category=cat) is None: db.categories.create(category=cat, description="Reject Probe") for a in (acct, other): code, out, err = cli_runner("account", "add", a, "Reject " + a) assert code == 0 code, out, err = cli_runner("txn", "post", acct, cat, "5.00", "2099-02-02", "seed") assert code == 0

before = counts() rejected = [ ("account", "add", acct, "Duplicate"), ("account", "add", "bad code", "Name"), ("account", "delete", acct), ("txn", "post", acct, cat, "1.234", "2099-02-03"), ("txn", "post", acct, cat, "1.00", "2026-02-30"), ("txn", "post", acct, "ZZNOCAT9", "1.00", "2099-02-03"), ("txn", "post", acct, cat, "0", "2099-02-03"), ("transfer", acct, acct, "1.00", "2099-02-03"), ("transfer", acct, other, "0.50", "2099-02-30"), ] for args in rejected: code, out, err = cli_runner(*args) print(args, code, err) assert code == 3, args assert out == "", args assert counts() == before === END AC contract-rejected-writes-change-nothing ===
